cmake_minimum_required(VERSION 3.16)
project(sort_aggregation_topk_fused LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ASCEND_CANN_PACKAGE_PATH)
    if(DEFINED ENV{ASCEND_CANN_PACKAGE_PATH})
        set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_CANN_PACKAGE_PATH}")
    elseif(DEFINED ENV{ASCEND_TOOLKIT_HOME})
        set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_TOOLKIT_HOME}")
    else()
        set(ASCEND_CANN_PACKAGE_PATH "/usr/local/Ascend/ascend-toolkit/latest")
    endif()
endif()
set(SOC_VERSION "Ascend910B1" CACHE STRING "SOC version")
if(EXISTS "${ASCEND_CANN_PACKAGE_PATH}/tikcpp/ascendc_kernel_cmake")
    set(ASCENDC_CMAKE_PATH "${ASCEND_CANN_PACKAGE_PATH}/tikcpp/ascendc_kernel_cmake")
elseif(EXISTS "${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/tikcpp/ascendc_kernel_cmake")
    set(ASCENDC_CMAKE_PATH "${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/tikcpp/ascendc_kernel_cmake")
elseif(EXISTS "${ASCEND_CANN_PACKAGE_PATH}/x86_64-linux/tikcpp/ascendc_kernel_cmake")
    set(ASCENDC_CMAKE_PATH "${ASCEND_CANN_PACKAGE_PATH}/x86_64-linux/tikcpp/ascendc_kernel_cmake")
else()
    file(GLOB_RECURSE ASCENDC_CMAKE_FILES "${ASCEND_CANN_PACKAGE_PATH}/../*/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
    if(ASCENDC_CMAKE_FILES)
        list(GET ASCENDC_CMAKE_FILES 0 FIRST_CMAKE)
        get_filename_component(ASCENDC_CMAKE_PATH "${FIRST_CMAKE}" DIRECTORY)
    else()
        message(STATUS "Cannot find ascendc_kernel_cmake; skipping device kernel build on host without toolkit")
    endif()
endif()
if(ASCENDC_CMAKE_PATH)
    include(${ASCENDC_CMAKE_PATH}/ascendc.cmake)
    ascendc_library(sort_aggregation_topk_fused_kernel_lib SHARED
        op_kernel/sort_aggregation_topk_fused_kernel.cpp)
    add_library(sort_aggregation_topk_fused_host SHARED
        op_host/sort_aggregation_topk_fused_host.cpp)
    target_include_directories(sort_aggregation_topk_fused_host PRIVATE
        ${ASCEND_CANN_PACKAGE_PATH}/include op_host)
    target_link_directories(sort_aggregation_topk_fused_host PRIVATE
        ${ASCEND_CANN_PACKAGE_PATH}/lib64)
    target_link_libraries(sort_aggregation_topk_fused_host PRIVATE
        ascendcl sort_aggregation_topk_fused_kernel_lib)
    add_library(sort_aggregation_topk_fused_op_def SHARED
        op_host/sort_aggregation_topk_fused_def.cpp)
    target_include_directories(sort_aggregation_topk_fused_op_def PRIVATE
        ${ASCEND_CANN_PACKAGE_PATH}/include)
    enable_testing()
    add_subdirectory(examples)
    install(TARGETS sort_aggregation_topk_fused_host LIBRARY DESTINATION lib)
    install(FILES op_host/sort_aggregation_topk_fused_host.h DESTINATION include)
endif()
