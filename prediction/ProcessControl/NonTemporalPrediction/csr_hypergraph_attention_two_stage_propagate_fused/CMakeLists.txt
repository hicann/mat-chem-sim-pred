cmake_minimum_required(VERSION 3.16)
project(csr_hypergraph_attention_two_stage_propagate_fused LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(NOT ASCEND_CANN_PACKAGE_PATH)
    if(DEFINED ENV{ASCEND_CANN_PACKAGE_PATH})
        set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_CANN_PACKAGE_PATH}")
    elseif(DEFINED ENV{ASCEND_TOOLKIT_HOME})
        set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_TOOLKIT_HOME}")
    else()
        set(ASCEND_CANN_PACKAGE_PATH "/usr/local/Ascend/ascend-toolkit/latest")
    endif()
endif()
set(SOC_VERSION "Ascend910B1" CACHE STRING "SOC version")
if(EXISTS "${ASCEND_CANN_PACKAGE_PATH}/tikcpp/ascendc_kernel_cmake")
    set(ASCENDC_CMAKE_PATH "${ASCEND_CANN_PACKAGE_PATH}/tikcpp/ascendc_kernel_cmake")
elseif(EXISTS "${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/tikcpp/ascendc_kernel_cmake")
    set(ASCENDC_CMAKE_PATH "${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/tikcpp/ascendc_kernel_cmake")
else()
    set(ASCENDC_CMAKE_PATH "${ASCEND_CANN_PACKAGE_PATH}/x86_64-linux/tikcpp/ascendc_kernel_cmake")
endif()
include(${ASCENDC_CMAKE_PATH}/ascendc.cmake)
ascendc_library(csr_hypergraph_attention_two_stage_propagate_fused_kernel_lib SHARED
    op_kernel/csr_hypergraph_attention_two_stage_propagate_fused_kernel.cpp)
add_library(csr_hypergraph_attention_two_stage_propagate_fused_host SHARED
    op_host/csr_hypergraph_attention_two_stage_propagate_fused_host.cpp)
target_include_directories(csr_hypergraph_attention_two_stage_propagate_fused_host PRIVATE
    ${ASCEND_CANN_PACKAGE_PATH}/include op_host)
target_link_directories(csr_hypergraph_attention_two_stage_propagate_fused_host PRIVATE
    ${ASCEND_CANN_PACKAGE_PATH}/lib64)
target_link_libraries(csr_hypergraph_attention_two_stage_propagate_fused_host PRIVATE
    ascendcl csr_hypergraph_attention_two_stage_propagate_fused_kernel_lib)
add_library(csr_hypergraph_attention_two_stage_propagate_fused_op_def SHARED
    op_host/csr_hypergraph_attention_two_stage_propagate_fused_def.cpp)
target_include_directories(csr_hypergraph_attention_two_stage_propagate_fused_op_def PRIVATE
    ${ASCEND_CANN_PACKAGE_PATH}/include)
enable_testing()
add_subdirectory(examples)
option(ENABLE_HOST_SANITIZERS "Enable ASAN and UBSAN for host and smoke code" OFF)
if(ENABLE_HOST_SANITIZERS)
    set(HOST_SANITIZER_LIBRARY_DIR "" CACHE PATH "Directory containing libasan.so and libubsan.so")
    if(NOT HOST_SANITIZER_LIBRARY_DIR)
        file(GLOB host_asan_candidates "/usr/lib/gcc/*/*/libasan.so")
        list(LENGTH host_asan_candidates host_asan_candidate_count)
        if(host_asan_candidate_count EQUAL 0)
            message(FATAL_ERROR "ENABLE_HOST_SANITIZERS needs libasan.so; set HOST_SANITIZER_LIBRARY_DIR")
        endif()
        list(GET host_asan_candidates 0 host_asan_library)
        get_filename_component(HOST_SANITIZER_LIBRARY_DIR "${host_asan_library}" DIRECTORY)
    endif()
    if(NOT EXISTS "${HOST_SANITIZER_LIBRARY_DIR}/libasan.so" OR NOT EXISTS "${HOST_SANITIZER_LIBRARY_DIR}/libubsan.so")
        message(FATAL_ERROR "ENABLE_HOST_SANITIZERS needs libasan.so and libubsan.so; set HOST_SANITIZER_LIBRARY_DIR")
    endif()
    foreach(target csr_hypergraph_attention_two_stage_propagate_fused_host
                   csr_hypergraph_attention_two_stage_propagate_fused_op_def
                   test_aclnn_csr_hypergraph_attention_two_stage_propagate_fused)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_directories(${target} PRIVATE ${HOST_SANITIZER_LIBRARY_DIR})
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endforeach()
endif()
install(TARGETS csr_hypergraph_attention_two_stage_propagate_fused_host LIBRARY DESTINATION lib)
install(FILES op_host/csr_hypergraph_attention_two_stage_propagate_fused_host.h DESTINATION include)
